import React, { useState, useEffect } from 'react';
import { ChevronDownIcon, SearchIcon, BarChart2Icon, XIcon, DownloadIcon, TrashIcon } from 'lucide-react';
import { predictStudentDropout, uploadDataset, getDatasets, deleteDataset, downloadDataset } from '../Api/internal';

function Dataset() {
  const [showPredictionPopup, setShowPredictionPopup] = useState(false);
  const [searchTerm, setSearchTerm] = useState('');
  const [sizeFilter, setSizeFilter] = useState('All Sizes');
  const [predictionResult, setPredictionResult] = useState(null);
  const [selectedFile, setSelectedFile] = useState(null);
  const [selectedModel, setSelectedModel] = useState('XGBoost');
  const [datasets, setDatasets] = useState([]);

  useEffect(() => {
    fetchDatasets();
  }, []);

  const fetchDatasets = async () => {
    const res = await getDatasets();
    setDatasets(res.data);
  };

  const handleFileChange = (e) => {
    if (e.target.files.length > 0) {
      setSelectedFile(e.target.files[0]);
    }
  };

  const handlePredict = async () => {
    if (!selectedFile) {
      alert('Please upload a CSV file.');
      return;
    }
    try {
      const result = await predictStudentDropout(selectedFile, selectedModel);
      setPredictionResult(result);
      setShowPredictionPopup(true);
    } catch (error) {
      alert('Error while predicting dropout risk.');
    }
  };

  const handleConfirmUpload = async () => {
    if (!selectedFile) {
      alert('No file selected.');
      return;
    }
    await uploadDataset(selectedFile);
    setShowPredictionPopup(false);
    fetchDatasets();
  };

  const handleDelete = async (id) => {
    await deleteDataset(id);
    fetchDatasets();
  };

  const handleDownload = async (id, name) => {
    const res = await downloadDataset(id);
    const url = window.URL.createObjectURL(new Blob([res.data]));
    const link = document.createElement("a");
    link.href = url;
    link.setAttribute("download", name);
    document.body.appendChild(link);
    link.click();
  };

  const filteredDatasets = datasets.filter(dataset => {
    const matchesName = dataset.name.toLowerCase().includes(searchTerm.toLowerCase());
    const matchesSize =
      sizeFilter === 'All Sizes' ||
      (sizeFilter === 'Small (<1MB)' && dataset.sizeBytes < 1024 * 1024) ||
      (sizeFilter === 'Medium (1-3MB)' && dataset.sizeBytes >= 1024 * 1024 && dataset.sizeBytes <= 3 * 1024 * 1024) ||
      (sizeFilter === 'Large (>3MB)' && dataset.sizeBytes > 3 * 1024 * 1024);
    return matchesName && matchesSize;
  });

  return (
    <div className="w-full min-h-screen bg-[#ffffff] p-4 sm:p-8">
      <div className="max-w-7xl mx-auto space-y-8">
        {/* Upload Section */}
        <div className="bg-white rounded-xl p-6 sm:p-8 shadow-lg">
          <h2 className="text-2xl font-semibold text-gray-800 mb-2">Upload New Dataset</h2>
          <input type="file" accept=".csv" onChange={handleFileChange} className="border rounded-md p-2 w-full" />
          <select value={selectedModel} onChange={(e) => setSelectedModel(e.target.value)} className="border rounded-md p-2 w-full mt-4">
            <option>XGBoost</option>
            <option>Gradient Boosting</option>
            <option>Random Forest</option>
            <option>Linear Regression</option>
          </select>
          <button onClick={handlePredict} className="bg-[#9078e2] text-white py-2 px-6 rounded-md mt-4 flex items-center">
            <BarChart2Icon size={18} className="mr-2" /> Predict
          </button>
        </div>

        {/* Manage Section */}
        <div className="bg-white rounded-xl p-6 sm:p-8 shadow-lg">
          <h2 className="text-2xl font-semibold mb-4">Manage Datasets</h2>
          <input type="text" placeholder="Search datasets..." className="border rounded-md p-2 w-full mb-4" value={searchTerm} onChange={(e) => setSearchTerm(e.target.value)} />
          <select value={sizeFilter} onChange={(e) => setSizeFilter(e.target.value)} className="border rounded-md p-2 w-full mb-4">
            <option>All Sizes</option>
            <option>Small (&lt;1MB)</option>
            <option>Medium (1-3MB)</option>
            <option>Large (&gt;3MB)</option>
          </select>

          <table className="min-w-full border">
            <thead>
              <tr className="bg-gray-100">
                <th className="p-2">Name</th>
                <th className="p-2">Size</th>
                <th className="p-2">Date</th>
                <th className="p-2">Actions</th>
              </tr>
            </thead>
            <tbody>
              {filteredDatasets.map(dataset => (
                <tr key={dataset._id} className="border-t">
                  <td className="p-2">{dataset.name}</td>
                  <td className="p-2">{dataset.size}</td>
                  <td className="p-2">{new Date(dataset.date).toLocaleDateString()}</td>
                  <td className="p-2 flex space-x-2">
                    <button onClick={() => handleDownload(dataset._id, dataset.name)} className="text-blue-600"><DownloadIcon size={16} /></button>
                    <button onClick={() => handleDelete(dataset._id)} className="text-red-600"><TrashIcon size={16} /></button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>

      {/* Popup */}
      {showPredictionPopup && predictionResult && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
          <div className="bg-white rounded-lg shadow-xl max-w-md w-full">
            <div className="p-6">
              <h3 className="text-lg font-medium mb-4">Prediction Results</h3>
              <table className="w-full mb-4">
                <tbody>
                  {Object.entries(predictionResult).map(([key, value], index) => (
                    <tr key={index}>
                      <td className="py-2 px-3">{key}</td>
                      <td className="py-2 px-3 text-right">{value}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
              <div className="flex justify-end gap-x-2">
                <button onClick={() => setShowPredictionPopup(false)} className="bg-[#c8c8ff] text-gray-700 py-2 px-6 rounded-md">Close</button>
                <button onClick={handleConfirmUpload} className="bg-[#9078e2] text-white py-2 px-4 rounded-md">Confirm & Upload</button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

export default Dataset;
